  function showToast(message) {
    const toast = document.getElementById('atelier-toast');

  // UPDATE HEADER CART COUNTERS
  function updateCartUI() {
    const totalCount = cart.reduce((sum, item) => sum + item.qty, 0);
    const totalPrice = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);

    // Update App Shell Cart Badges if present
    const headerCartBadge = document.querySelector('button[aria-label="Shopping bag"] span:last-child');
    if (headerCartBadge) {
      headerCartBadge.textContent = totalCount;
    }

    document.getElementById('cart-count-display').textContent = totalCount;
    document.getElementById('cart-subtotal').innerHTML = `${totalPrice.toFixed(3)} <span class="text-secondary text-xs uppercase">KWD</span>`;

    const container = document.getElementById('cart-items-container');
    const emptyState = document.getElementById('cart-empty-state');

    if (cart.length === 0) {
      container.innerHTML = '';
      container.appendChild(emptyState);
      emptyState.classList.remove('hidden');
    } else {
      emptyState.classList.add('hidden');
      container.innerHTML = '';
      cart.forEach((item, index) => {
        const itemEl = document.createElement('div');
        itemEl.className = 'flex items-center gap-space-sm p-space-sm bg-surface';
        itemEl.innerHTML = `
          <img src="${item.image}" alt="${item.name}" class="w-16 h-16 object-cover bg-surface-container">
          <div class="flex flex-col flex-grow">
            <span class="font-headline-sm text-[14px] text-primary line-clamp-1">${item.name}</span>
            <span class="font-price-sm text-price-sm text-secondary">${item.price.toFixed(3)} KWD</span>
            <div class="flex items-center gap-2 mt-1">
              <button onclick="changeQty(${index}, -1)" class="w-6 h-6 bg-surface-container flex items-center justify-center text-xs text-on-surface hover:bg-secondary hover:text-white">-</button>
              <span class="text-xs font-semibold px-1">${item.qty}</span>
              <button onclick="changeQty(${index}, 1)" class="w-6 h-6 bg-surface-container flex items-center justify-center text-xs text-on-surface hover:bg-secondary hover:text-white">+</button>
            </div>
          </div>
          <button onclick="removeCartItem(${index})" class="text-outline hover:text-error p-1" title="Remove">
            <span class="material-symbols-outlined text-[20px]">delete</span>
          </button>
        `;
        container.appendChild(itemEl);
      });
    }
  }

  window.changeQty = function(index, delta) {
    if (cart[index]) {
      cart[index].qty += delta;
      if (cart[index].qty <= 0) {
        cart.splice(index, 1);
      }
      updateCartUI();
    }
  };

  window.removeCartItem = function(index) {
    cart.splice(index, 1);
    updateCartUI();
    showToast('Item removed from bag');
  };

  function addToCart(productId) {
    const prod = catalog[productId];
    if (!prod) return;

    const existingIndex = cart.findIndex(item => item.id === productId);
    if (existingIndex > -1) {
      cart[existingIndex].qty += 1;
    } else {
      cart.push({
        id: prod.id,
        name: prod.title,
        price: prod.price,
        image: prod.image,
        qty: 1
      });
    }

    updateCartUI();
    showToast(`Added: ${prod.title}`);
    openCartDrawer();
  }
